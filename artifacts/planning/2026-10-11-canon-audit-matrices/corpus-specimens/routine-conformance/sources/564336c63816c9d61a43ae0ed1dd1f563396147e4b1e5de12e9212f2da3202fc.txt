id: adaptive-pass
version: 1.4.0
name: Adaptive Depth Pass
description: Cost-minimizing depth escalation.
variables:
  reads:
    - target
  writes:
    - name: adaptive_signal_quality
      type: string
      description: "For adaptive mode: signal quality assessment at current stage (adequate or insufficient)"
    - name: all_artifact_paths
      type: array
      description: Accumulated list of all artifact paths across all units
steps:
  - kind: technique
    id: run-stage-1
    technique:
      name: adaptive-analysis::stage-1-sdl
      inputs:
        target_content: "{target}"
  - kind: technique
    id: run-stage-2
    technique:
      name: adaptive-analysis::stage-2-l12
      inputs:
        target_content: "{target}"
    when: adaptive_signal_quality == 'insufficient'
  - kind: technique
    id: run-stage-3
    technique:
      name: adaptive-analysis::stage-3-full
      inputs:
        target_content: "{target}"
    when: adaptive_signal_quality == 'insufficient'
  - kind: action
    id: collect-paths
    actions:
      - action: set
        target: all_artifact_paths
        description: Accumulated artifact paths across the run.
exits:
  - id: done
    isDefault: true
