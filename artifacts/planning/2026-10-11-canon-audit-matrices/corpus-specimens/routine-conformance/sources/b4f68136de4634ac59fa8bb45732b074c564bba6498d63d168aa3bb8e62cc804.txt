id: implementation-analysis
version: 2.16.0
name: Implementation Analysis
description: Analyze the current implementation to understand effectiveness, establish baselines, and identify opportunities for improvement.
variables:
  reads:
    - assumption_outcome
    - base_branch
    - branch_name
    - component_git_dir
    - component_name
    - is_review_mode
    - planning_folder_path
    - pr_number
    - problem_statement
    - requirements
    - target_path
    - target_repo
  writes:
    - name: assumption_review_presentation
      type: object
      description: Judgement-augmentation context per open assumption — decision space, non-resolvability rationale, technical context, agent's position, reversibility — with a link to the assumptions log.
    - name: assumptions_log
      type: object
      description: The assumptions log — every assumption with its category, agent position and resolution status, and the record of truth for assumption outcomes.
    - name: changed_files
      type: array
      description: Ordered list of changed file paths (each `.filename` from the pulls files endpoint).
      defaultValue: []
    - name: has_deferred_assumptions
      type: boolean
      description: Whether any assumption was deferred to stakeholders.
      defaultValue: false
    - name: has_open_assumptions
      type: boolean
      description: Whether unresolved assumptions requiring stakeholder input remain.
      defaultValue: false
    - name: has_resolvable_assumptions
      type: boolean
      description: Whether code-resolvable assumptions exist that can be validated through targeted analysis.
      defaultValue: false
    - name: open_assumptions
      type: array
      description: The residual open assumptions awaiting a stakeholder judgement.
required: false
techniques:
  - scatter-gather
steps:
  - kind: technique
    id: review-baseline-state
    technique: review-baseline-state
    when: is_review_mode == true
  - kind: technique
    id: analyze-implementation
    technique: analyze
  - kind: technique
    id: collect-assumptions
    technique:
      name: review-assumptions::collect
      inputs:
        assumption_categories: Current Behavior, Gap Identification, Baseline Interpretation, Dependency Understanding
  - kind: technique
    id: document
    technique: document
  - kind: technique
    id: update-assumptions-log
    technique: review-assumptions::record
  - kind: routine
    id: assumption-convergence
    routine: converge-assumptions
    outputs:
      assumptions_log: assumptions_log
      has_resolvable_assumptions: has_resolvable_assumptions
      has_open_assumptions: has_open_assumptions
      open_assumptions: open_assumptions
  - kind: technique
    id: present-resolved-assumptions
    technique: review-assumptions::assemble-open-set
    actions:
      - action: message
        message: |-
          **Assumptions**

          {assumption_review_presentation}
exits:
  - id: done
    isDefault: true
outcome:
  - Current state documented with baseline metrics
  - Gaps identified and prioritized
  - Success criteria defined with measurement methodology
