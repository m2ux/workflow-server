id: synthesis-pass
version: 3.0.0
name: Synthesis Pass
description: Run the synthesis lens against each full-prism unit's structural and adversarial artifacts.
variables:
  reads:
    - all_artifact_paths
    - analysis_units
  writes:
    - name: all_artifact_paths
      type: array
      description: Accumulated list of all artifact paths across all units
    - name: current_unit
      type: object
      description: Current analysis unit during iteration loop
techniques:
  - scatter-gather
steps:
  - kind: routine
    id: synthesis
    routine: per-unit-pass
    with:
      pass_operation: full-prism::synthesis
      prior_artifact_paths: all_artifact_paths
      unit_pipeline_mode: full-prism
    outputs:
      all_artifact_paths: all_artifact_paths
      current_unit: current_unit
exits:
  - id: done
    isDefault: true
