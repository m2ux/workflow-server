id: positive-case
version: 1.0.0
name: Positive Case
description: Refer to the guarded rename over a symbol the graph holds, with the gate accepting the edit list.
variables:
  reads:
    - positive_file_path
    - positive_new_name
    - positive_symbol_name
  writes:
    - name: positive_changes
      type: array
      description: The per-file edit list, each edit carrying the confidence its provenance earns.
    - name: positive_rename_approved
      type: boolean
      description: Whether the previewed edit list was accepted at the gate.
      defaultValue: true
    - name: positive_change_report
      type: object
      description: The symbols and execution flows the applied rename actually moved, read back off the diff.
required: true
steps:
  - kind: routine
    id: rename-held-symbol
    routine: gitnexus::guarded-rename
    with:
      symbol_name: "{positive_symbol_name}"
      new_name: "{positive_new_name}"
      file_path: "{positive_file_path}"
    outputs:
      changes: positive_changes
      rename_approved: positive_rename_approved
      change_report: positive_change_report
exits:
  - id: gated
    isDefault: true
outcome:
  - The preview landed an edit list for a symbol the graph holds, the spliced gate accepted it, the rename wrote the edits and the diff read back what moved
