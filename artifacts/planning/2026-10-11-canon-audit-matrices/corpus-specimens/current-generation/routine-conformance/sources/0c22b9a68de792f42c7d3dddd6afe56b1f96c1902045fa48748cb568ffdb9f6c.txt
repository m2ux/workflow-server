id: assumptions-review
version: 2.16.0
name: Assumptions Review
description: Converge open assumptions via analyse-challenge, then batch residual stakeholder judgements.
variables:
  reads:
    - branch_name
    - component_git_dir
    - is_review_mode
    - issue_number
    - issue_platform
    - needs_comprehension
    - needs_further_discussion
    - needs_plan_revision
    - planning_folder_path
    - pr_number
    - problem_statement
    - requirements
    - stealth_mode
    - target_path
    - target_repo
  writes:
    - name: assumption_outcome
      type: string
      description: The outcome given the assumption under discussion. Empty before one has been decided.
      defaultValue: ""
    - name: assumption_review_presentation
      type: object
      description: Judgement-augmentation context per open assumption — decision space, non-resolvability rationale, technical context, agent's position, reversibility — with a link to the assumptions log.
    - name: assumptions_log
      type: object
      description: The assumptions log — every assumption with its category, agent position and resolution status, and the record of truth for assumption outcomes.
    - name: deferred_items_register
      type: string
      description: The out-of-scope deferrals register as it stands after this activity's rows.
      defaultValue: ""
    - name: has_deferred_assumptions
      type: boolean
      description: Whether any assumption was deferred to stakeholders.
      defaultValue: false
    - name: has_open_assumptions
      type: boolean
      description: Whether unresolved assumptions requiring stakeholder input remain.
      defaultValue: false
    - name: has_resolvable_assumptions
      type: boolean
      description: Whether code-resolvable assumptions exist that can be validated through targeted analysis.
      defaultValue: false
    - name: open_assumptions
      type: array
      description: The residual open assumptions awaiting a stakeholder judgement.
    - name: post_summary_approved
      type: boolean
      description: Whether the deferred-assumptions summary is posted to the issue tracker, as decided at the post-summary-review gate.
      defaultValue: false
required: true
techniques:
  - scatter-gather
steps:
  - kind: technique
    id: evaluate-open-assumptions
    technique: review-assumptions::collect
  - kind: routine
    id: assumption-convergence
    routine: converge-assumptions
    outputs:
      assumptions_log: assumptions_log
      has_resolvable_assumptions: has_resolvable_assumptions
      has_open_assumptions: has_open_assumptions
      open_assumptions: open_assumptions
  - kind: technique
    id: present-residual-assumptions
    technique: review-assumptions::assemble-open-set
    when: is_review_mode != true
  - kind: routine
    id: residual-assumptions
    routine: residual-assumption-interview
    outputs:
      assumption_outcome: assumption_outcome
      has_deferred_assumptions: has_deferred_assumptions
  - kind: technique
    id: update-assumptions-log
    technique: review-assumptions::record
  - kind: technique
    id: register-deferred-assumptions
    technique: manage-registers::append-deferred-item
  - kind: checkpoint
    id: post-summary-review
    condition:
      type: and
      conditions:
        - type: simple
          variable: stealth_mode
          operator: "!="
          value: true
        - type: simple
          variable: is_review_mode
          operator: "!="
          value: true
        - type: simple
          variable: issue_platform
          operator: exists
        - type: simple
          variable: has_deferred_assumptions
          operator: ==
          value: true
    message: Assumptions were deferred and need attention from stakeholders on {issue_platform}.
    options:
      - id: post-summary
        label: Post summary to issue tracker
        description: Post the resolution summary
        effect:
          setVariable:
            post_summary_approved: true
      - id: skip-posting
        label: Skip posting
        description: Do not post to issue tracker
        effect:
          setVariable:
            post_summary_approved: false
  - kind: technique
    id: post-summary-jira
    technique: atlassian::comment-jira-issue
    when: stealth_mode != true && is_review_mode != true && issue_platform == 'jira' && has_deferred_assumptions == true && post_summary_approved == true
  - kind: technique
    id: post-summary-github
    technique:
      name: github::comment-issue
      inputs:
        repo_path: component_git_dir
    when: stealth_mode != true && is_review_mode != true && issue_platform == 'github' && has_deferred_assumptions == true && post_summary_approved == true
exits:
  - id: needs-comprehension
    when: needs_comprehension == true
  - id: needs-plan-revision
    when: needs_plan_revision == true
  - id: needs-further-discussion
    when: needs_further_discussion == true
  - id: review-mode
    when: is_review_mode == true
  - id: assumptions-approved
    isDefault: true
outcome:
  - Open assumptions converged through analyse-challenge before any residual ask
  - Residual stakeholder judgements batched (or individually reviewed only when requested)
  - Implementation can proceed on a vetted set of assumption decisions, with any deferred items explicitly tracked
  - External stakeholders are notified of deferred assumptions that need their attention
