id: harvest-debt-and-report
version: 1.3.0
name: Harvest Debt and Report
description: Harvest the deliberate-simplification markers into a debt ledger and, when markers exist, append an honesty-bounded gain scoreboard.
variables:
  reads:
    - artifact_dir
    - task_description
  writes:
    - name: has_debt_markers
      type: boolean
      description: Whether the harvest found any ponytail marker
      defaultValue: false
steps:
  - kind: technique
    id: harvest-markers
    technique: harvest-debt
  - kind: technique
    id: verify-artifact-conformance
    technique:
      name: verify-artifact-conforms
      inputs:
        artifact_dir: artifact_dir
        guide_map: ponytail/README#planning-artifact-to-guide-map
  - kind: technique
    id: report-gain
    technique: report-gain
    when: has_debt_markers == true
    actions:
      - action: message
        message: |-
          Honest gain from the deliberate simplifications recorded in this pass:

          {gain_scoreboard}
outcome:
  - Every deliberate simplification is recorded as trackable debt with its upgrade trigger
  - Where the tree carries any such debt, the ledger also states what the pass can honestly claim for it
